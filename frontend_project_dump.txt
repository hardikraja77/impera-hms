import { BrowserRouter, Routes, Route } from 'react-router-dom';

import ProtectedRoute from './components/ProtectedRoute';
import HomePage from './pages/HomePage';
import LoginPage from './pages/LoginPage';
import AdminDashboard from './pages/admin/AdminDashboard';
import DoctorDashboard from './pages/doctor/DoctorDashboard';
import ReceptionistDashboard from './pages/receptionist/ReceptionistDashboard';

function App() {
  return (
    <AuthProvider>
      <EchoInitializer />
      <BrowserRouter>
        <Routes>
          {/* Public Routes */}
          <Route path="/" element={<HomePage />} />

          {/* Login Routes */}
          <Route path="/admin-login" element={<LoginPage role="admin" />} />
          <Route path="/doctor-login" element={<LoginPage role="doctor" />} />

              <ProtectedRoute allowedRoles={['admin']}>
                <AdminDashboard />
              </ProtectedRoute>
            }

          <Route

            }
          />

          {/* 404 Fallback */}
          <Route path="*" element={<NotFoundPage />} />

      </BrowserRouter>

}

export default App;

import axios from 'axios';

const api = axios.create({
  baseURL: '/api',
  headers: {
    'Content-Type': 'application/json',
    'Accept': 'application/json',
  },
  timeout: 15000,
});

// Attach token to every request
api.interceptors.request.use((config) => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Global response error handler
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response) {
      const { status, data } = error.response;

      // Unauthorized — token expired or invalid
      if (status === 401) {
        localStorage.removeItem('token');
        localStorage.removeItem('user');
        localStorage.removeItem('tokenExpiresAt');
        window.location.href = '/';
        return Promise.reject(error);
      }

      // Forbidden — role mismatch
      if (status === 403) {
        console.warn('[API] Access denied:', data?.message);
      }

      // Not found — missing endpoint or resource
      if (status === 404) {
        console.error('[API] Resource not found:', error.config?.url, data?.message);
      }

      // Server errors
      if (status >= 500) {
        console.error('[API] Server error:', error.config?.url, data?.message);
      }
    } else if (error.code === 'ECONNABORTED') {
      console.error('[API] Request timeout:', error.config?.url);
    } else {
      console.error('[API] Network error:', error.message);
    }

    return Promise.reject(error);
  }
);

/**
 * Get a user-friendly error message from an API error
 */
export function getErrorMessage(error) {
  if (error.response?.data?.message) {
    return error.response.data.message;
  }
  if (error.response?.data?.errors) {
    const errors = error.response.data.errors;
    const firstKey = Object.keys(errors)[0];
    if (firstKey) {
      const val = errors[firstKey];
      return Array.isArray(val) ? val[0] : val;
    }
  }
  if (error.code === 'ECONNABORTED') return 'Request timed out. Please try again.';
  if (!error.response) return 'Network error. Please check your connection.';
  return 'Something went wrong. Please try again.';
}

/**
 * Generate a WhatsApp deep link (works on mobile and desktop)
 */
export function generateWhatsAppLink(phone, message) {
  const cleaned = phone.replace(/[^0-9]/g, '');
  const encoded = encodeURIComponent(message);
  return `https://wa.me/${cleaned}?text=${encoded}`;
}

export default api;

import Echo from 'laravel-echo';
import Pusher from 'pusher-js';

// Make Pusher available globally for Echo
window.Pusher = Pusher;

let echo = null;

/**
 * Initialize Laravel Echo connection
 * Only connects when a valid token exists (user is authenticated)
 */
export function initEcho() {
  const token = localStorage.getItem('token');
  if (!token) return null;
  if (echo) return echo;

  echo = new Echo({
    broadcaster: 'reverb',
    key: import.meta.env.VITE_REVERB_APP_KEY,
    wsHost: import.meta.env.VITE_REVERB_HOST || 'localhost',
    wsPort: import.meta.env.VITE_REVERB_PORT || 8080,
    wssPort: import.meta.env.VITE_REVERB_PORT || 8080,
    forceTLS: (import.meta.env.VITE_REVERB_SCHEME || 'http') === 'https',
    enabledTransports: ['ws', 'wss'],
    auth: {
      headers: {
        Authorization: `Bearer ${token}`,
        Accept: 'application/json',
      },
    },
    authEndpoint: '/api/broadcasting/auth',
  });

  return echo;
}

/**
 * Get the Echo instance
 */
export function getEcho() {
  return echo;
}

/**
 * Disconnect Echo and clean up
 */
export function disconnectEcho() {
  if (echo) {
    echo.disconnect();
    echo = null;
  }
}

/**
 * Subscribe to a private channel
 */
export function subscribePrivate(channelName, event, callback) {
  const e = initEcho();
  if (!e) return null;

  return e.private(channelName).listen(event, callback);
}

export default echo;

import { Navigate, useLocation } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

const ROLE_DASHBOARDS = {
  admin: '/admin',
  doctor: '/doctor',
  receptionist: '/receptionist',
};

const ROLE_LOGIN = {
  admin: '/admin-login',
  doctor: '/doctor-login',
  receptionist: '/receptionist-login',
};

export default function ProtectedRoute({ children, allowedRoles }) {
  const { user, loading } = useAuth();
  const location = useLocation();

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-50">
        <div className="text-center">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600 mx-auto"></div>
          <p className="mt-4 text-gray-600">Loading...</p>
        </div>
      </div>
    );
  }

  // Not logged in — redirect to home
  if (!user) {
    return <Navigate to="/" state={{ from: location }} replace />;
  }

  // Logged in but wrong role — show access denied with option to go to correct dashboard
  if (allowedRoles && !allowedRoles.includes(user.role)) {
    const correctDashboard = ROLE_DASHBOARDS[user.role] || '/';
    const correctLogin = ROLE_LOGIN[user.role] || '/';

    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-50">
        <div className="max-w-md w-full bg-white rounded-xl shadow-lg p-8 text-center">
          <div className="text-6xl mb-4">🚫</div>
          <h1 className="text-2xl font-bold text-gray-900 mb-2">Access Denied</h1>
          <p className="text-gray-600 mb-6">
            You don't have permission to access this page.
            <br />
            <span className="text-sm text-gray-500">
              You are logged in as <strong className="capitalize">{user.role}</strong> ({user.email})
            </span>
          </p>
          <div className="space-y-3">
            <a
              href={correctDashboard}
              className="block w-full px-4 py-3 bg-blue-600 text-white rounded-lg font-medium hover:bg-blue-700 transition-colors"
            >
              Go to {user.role.charAt(0).toUpperCase() + user.role.slice(1)} Dashboard
            </a>
            <a
              href="/"
              className="block w-full px-4 py-3 bg-gray-100 text-gray-700 rounded-lg font-medium hover:bg-gray-200 transition-colors"
            >
              Back to Home
            </a>
          </div>
        </div>
      </div>
    );
  }

  return children;
}

import { createContext, useContext, useState, useEffect, useCallback } from 'react';
import api from '../api';

const AuthContext = createContext(null);

const ROLE_DASHBOARDS = {
  admin: '/admin',
  doctor: '/doctor',
  receptionist: '/receptionist',
};

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [tokenExpiresAt, setTokenExpiresAt] = useState(null);

  useEffect(() => {
    const token = localStorage.getItem('token');
    const expiresAt = localStorage.getItem('tokenExpiresAt');

    if (token) {
      // Check if token is expired
      if (expiresAt && new Date(expiresAt) < new Date()) {
        // Token expired, clear storage
        localStorage.removeItem('token');
        localStorage.removeItem('user');
        localStorage.removeItem('tokenExpiresAt');
        setLoading(false);
        return;
      }

      setTokenExpiresAt(expiresAt);

      api.get('/user')
        .then((res) => {
          setUser(res.data);
        })
        .catch(() => {
          localStorage.removeItem('token');
          localStorage.removeItem('user');
          localStorage.removeItem('tokenExpiresAt');
        })
        .finally(() => setLoading(false));
    } else {
      setLoading(false);
    }
  }, []);

  const login = async (email, password, role) => {
    const res = await api.post('/login', { email, password, role });
    localStorage.setItem('token', res.data.token);
    localStorage.setItem('user', JSON.stringify(res.data.user));

    if (res.data.expires_at) {
      localStorage.setItem('tokenExpiresAt', res.data.expires_at);
      setTokenExpiresAt(res.data.expires_at);
    }

    setUser(res.data.user);
    return res.data;
  };

  const logout = async () => {
    try {
      await api.post('/logout');
    } catch {
      // ignore logout errors
    }
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    localStorage.removeItem('tokenExpiresAt');
    setTokenExpiresAt(null);
    setUser(null);
  };

  const logoutAll = async () => {
    try {
      await api.post('/logout-all');
    } catch {
      // ignore
    }
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    localStorage.removeItem('tokenExpiresAt');
    setTokenExpiresAt(null);
    setUser(null);
  };

  // Check if token is about to expire (within 10 minutes)
  const isTokenExpiringSoon = useCallback(() => {
    if (!tokenExpiresAt) return false;
    const expiry = new Date(tokenExpiresAt);
    const now = new Date();
    const tenMinutes = 10 * 60 * 1000;
    return expiry - now < tenMinutes && expiry - now > 0;
  }, [tokenExpiresAt]);

  const getDashboardUrl = () => {
    if (!user) return '/';
    return ROLE_DASHBOARDS[user.role] || '/';
  };

  return (
    <AuthContext.Provider value={{
      user,
      login,
      logout,
      logoutAll,
      loading,
      getDashboardUrl,
      tokenExpiresAt,
      isTokenExpiringSoon,
    }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

import { useEffect, useRef } from 'react';
import { initEcho, disconnectEcho } from '../echo';
import { useAuth } from '../context/AuthContext';

/**
 * React hook to subscribe to a private Echo channel.
 * Automatically connects on mount and disconnects on unmount.
 *
 * @param {string} channel - The private channel name (e.g., 'doctor.1')
 * @param {string} event - The event name to listen for (e.g., '.patient.created')
 * @param {function} callback - Handler for the event data
 */
export function useEchoChannel(channel, event, callback) {
  const { user } = useAuth();
  const listenerRef = useRef(null);

  useEffect(() => {
    if (!user || !channel || !event) return;

    const echo = initEcho();
    if (!echo) return;

    try {
      const listener = echo.private(channel).listen(event, callback);
      listenerRef.current = listener;
    } catch (err) {
      console.error(`[Echo] Failed to subscribe to ${channel}:${event}`, err);
    }

    return () => {
      if (listenerRef.current) {
        try {
          echo.private(channel).stopListening(event);
        } catch {
          // ignore cleanup errors
        }
        listenerRef.current = null;
      }
    };
  }, [user, channel, event, callback]);

  return listenerRef;
}

/**
 * React hook to connect/disconnect Echo based on auth state
 */
export function useEchoConnection() {
  const { user, tokenExpiresAt } = useAuth();

  useEffect(() => {
    if (user) {
      initEcho();
    }

    return () => {
      disconnectEcho();
    };
  }, [user, tokenExpiresAt]);
}

export default useEchoChannel;

function validatePassword(password) {

  const [modalType, setModalType] = useState('');
  const [editItem, setEditItem] = useState(null);
  const [formData, setFormData] = useState({});
  const [modalErrors, setModalErrors] = useState({});
  const [saving, setSaving] = useState(false);
  const [deletingId, setDeletingId] = useState(null);

  // ── All fetch functions declared BEFORE hooks that reference them ──

  const fetchDoctors = useCallback(async () => {
    try {
      const res = await api.get('/admin/doctors');
      setDoctors(res.data);
    } catch (err) {
      console.error('[Admin] Doctors fetch failed:', getErrorMessage(err));
    }
  }, []);

  const fetchAnalytics = useCallback(async () => {
    setLoading(true);
    setError('');
    try {
      const params = new URLSearchParams();
      params.append('from_date', fromDate);
      params.append('to_date', toDate);
      if (analyticsDoctor) params.append('doctor_id', analyticsDoctor);
      if (analyticsDept) params.append('department', analyticsDept);
      const res = await api.get(`/admin/analytics?${params.toString()}`);
      setAnalytics(res.data);
    } catch (err) {
      console.error('[Admin] Analytics fetch failed:', getErrorMessage(err));
      setError(getErrorMessage(err));
    } finally {
      setLoading(false);
    }
  }, [fromDate, toDate, analyticsDoctor, analyticsDept]);

  const fetchReceptionists = useCallback(async () => {
    try {
      const res = await api.get('/admin/receptionists');
      setReceptionists(res.data);
    } catch (err) {
      console.error('[Admin] Receptionists fetch failed:', getErrorMessage(err));
    }
  }, []);

  const fetchPatients = useCallback(async () => {
    setLoading(true);
    setError('');
    try {
      const params = new URLSearchParams();
      if (search) params.append('search', search);
      if (filterStatus) params.append('status', filterStatus);
      if (filterDoctor) params.append('doctor_id', filterDoctor);
      if (filterDate) params.append('date', filterDate);
      const res = await api.get(`/admin/patients?${params.toString()}`);
      setPatients(res.data.data || []);
    } catch (err) {
      console.error('[Admin] Patients fetch failed:', getErrorMessage(err));
      setError(getErrorMessage(err));
    } finally {
      setLoading(false);
    }
  }, [search, filterStatus, filterDoctor, filterDate]);

  const fetchNextAppointments = useCallback(async () => {
    setLoadingNext(true);
    try {
      const params = new URLSearchParams();
      if (nextApptDate) params.append('date', nextApptDate);
      if (nextApptDoctor) params.append('doctor_id', nextApptDoctor);
      const res = await api.get(`/admin/next-appointments?${params.toString()}`);
      setNextAppointments(res.data || []);
    } catch (err) {
      console.error('[Admin] Next appointments fetch failed:', getErrorMessage(err));
      setError(getErrorMessage(err));
    } finally {
      setLoadingNext(false);
    }
  }, [nextApptDate, nextApptDoctor]);

  // ── Effects (now safe — all referenced functions are declared above) ──

  useEffect(() => {
    fetchDoctors();
  }, [fetchDoctors]);

  useEffect(() => {
    if (activeTab === 'analytics') fetchAnalytics();
    if (activeTab === 'doctors') fetchDoctors();
    if (activeTab === 'receptionists') fetchReceptionists();
    if (activeTab === 'patients') fetchPatients();
    if (activeTab === 'nextAppointments') fetchNextAppointments();
  }, [activeTab, fetchAnalytics, fetchDoctors, fetchReceptionists, fetchPatients, fetchNextAppointments]);

  useEffect(() => {
    if (activeTab === 'patients') fetchPatients();
  }, [search, filterStatus, filterDoctor, filterDate, fetchPatients, activeTab]);

  useEffect(() => {
    if (activeTab === 'analytics') fetchAnalytics();
  }, [fromDate, toDate, analyticsDoctor, analyticsDept, fetchAnalytics, activeTab]);

  useEffect(() => {
    if (activeTab === 'nextAppointments') fetchNextAppointments();
  }, [nextApptDate, nextApptDoctor, activeTab, fetchNextAppointments]);

  // ── Real-time Echo subscriptions for admin dashboard ──
  useEchoChannel(
    'admin.dashboard',
    '.patient.created',
    useCallback((e) => {
      console.log('[Echo] Admin: New patient created:', e);
      if (activeTab === 'analytics') fetchAnalytics();
      if (activeTab === 'patients') fetchPatients();
    }, [activeTab, fetchAnalytics, fetchPatients])
  );

  useEchoChannel(
    'admin.dashboard',
    '.patient.status.updated',
    useCallback((e) => {
      console.log('[Echo] Admin: Patient status updated:', e);
      if (activeTab === 'analytics') fetchAnalytics();
      if (activeTab === 'patients') fetchPatients();
    }, [activeTab, fetchAnalytics, fetchPatients])
  );

  useEchoChannel(
    'admin.dashboard',
    '.consultation.updated',
    useCallback((e) => {
      console.log('[Echo] Admin: Consultation updated:', e);
      if (activeTab === 'analytics') fetchAnalytics();
      if (activeTab === 'patients') fetchPatients();
      if (activeTab === 'nextAppointments') fetchNextAppointments();
    }, [activeTab, fetchAnalytics, fetchPatients, fetchNextAppointments])
  );

  // ── Modal & CRUD handlers ──

  const openModal = (type, item = null) => {
    setModalType(type);
    setEditItem(item);
    if (type === 'doctor') {
      setFormData(item ? {
        name: item.user?.name || '',
        email: item.user?.email || '',
        phone: item.phone || '',
        specialization: item.specialization || '',
        consultation_fee: item.consultation_fee || 0,
        available_days: item.available_days || [],
        available_from: item.available_from || '09:00',
        available_to: item.available_to || '17:00',
      } : {
        name: '', email: '', phone: '', password: '', password_confirmation: '',
        specialization: '', consultation_fee: 0,
        available_days: ['Mon','Tue','Wed','Thu','Fri'],
        available_from: '09:00', available_to: '17:00',
      });
    } else if (type === 'receptionist') {
      setFormData(item ? {
        name: item.name || '', email: item.email || '', phone: item.phone || '',
      } : { name: '', email: '', phone: '', password: '', password_confirmation: '' });
    }
    setModalErrors({});
    setShowModal(true);
  };

  const validateModal = () => {
    const errs = {};
    if (!formData.name?.trim()) errs.name = 'Name is required';
    if (!formData.email?.trim()) errs.email = 'Email is required';
    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email)) errs.email = 'Enter a valid email';
    if (modalType === 'doctor' && !formData.specialization?.trim()) errs.specialization = 'Specialization is required';
    if (!editItem) {
      if (!formData.password) errs.password = 'Password is required';
      else if (formData.password.length < 6) errs.password = 'Password must be at least 6 characters';
      if (formData.password !== formData.password_confirmation) errs.password_confirmation = 'Passwords do not match';
    }
    setModalErrors(errs);
    return Object.keys(errs).length === 0;
  };

  const handleSave = async () => {
    if (!validateModal()) return;
    setSaving(true);
    try {
      if (modalType === 'doctor') {
        if (editItem) {
          await api.put(`/admin/doctors/${editItem.id}`, formData);
        } else {
          await api.post('/admin/doctors', formData);
        }
        fetchDoctors();
      } else if (modalType === 'receptionist') {
        if (editItem) {
          await api.put(`/admin/receptionists/${editItem.id}`, formData);
        } else {
          await api.post('/admin/receptionists', formData);
        }
        fetchReceptionists();
      }
      setShowModal(false);
    } catch (err) {
      const serverErrors = err.response?.data?.errors;
      if (serverErrors) {
        const mapped = {};
        for (const [key, val] of Object.entries(serverErrors)) {
          mapped[key] = Array.isArray(val) ? val[0] : val;
        }
        setModalErrors(mapped);
      } else {
        setModalErrors({ _general: getErrorMessage(err) });
      }
    } finally {
      setSaving(false);
    }
  };

  const handleDelete = async (type, id) => {
    if (!confirm('Are you sure you want to delete this?')) return;
    setDeletingId(id);
    try {
      if (type === 'doctor') {
        await api.delete(`/admin/doctors/${id}`);
        fetchDoctors();
      } else {
        await api.delete(`/admin/receptionists/${id}`);
        fetchReceptionists();
      }
    } catch (err) {
      alert(getErrorMessage(err));
    } finally {
      setDeletingId(null);
    }
  };

  const handleWhatsAppEnquiry = (patient) => {
    const hospitalName = 'HealthCare';
    const doctorName = patient.doctor?.user?.name || 'Unassigned';
    const appointmentDate = new Date(patient.created_at).toLocaleDateString('en-IN');
    const appointmentTime = patient.consultation?.appointment_time
      ? new Date(patient.consultation.appointment_time).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
      : 'Not scheduled';

    const message = `Hello, this is regarding your appointment at ${hospitalName}.\n\nPatient: ${patient.name}\nDoctor: ${doctorName}\nScheduled Date: ${appointmentDate}\nTime: ${appointmentTime}\n\nPlease confirm your availability.`;

    const url = generateWhatsAppLink(patient.phone, message);
    window.open(url, '_blank');
  };

  // ── Navigation & display helpers ──

  const navItems = [
    { id: 'analytics', label: 'Analytics', icon: '📊', active: activeTab === 'analytics', onClick: () => setActiveTab('analytics') },
    { id: 'doctors', label: 'Doctors', icon: '👨‍⚕️', active: activeTab === 'doctors', onClick: () => setActiveTab('doctors'), badge: doctors.length },
    { id: 'receptionists', label: 'Receptionists', icon: '🧑‍💼', active: activeTab === 'receptionists', onClick: () => setActiveTab('receptionists'), badge: receptionists.length },
    { id: 'nextAppointments', label: 'Next Appointments', icon: '📅', active: activeTab === 'nextAppointments', onClick: () => setActiveTab('nextAppointments'), badge: nextAppointments.length },
    { id: 'patients', label: 'All Patients', icon: '👥', active: activeTab === 'patients', onClick: () => setActiveTab('patients') },
  ];

  const statusColors = {
    pending: 'bg-yellow-50 text-yellow-700 border-yellow-200',
    enquired: 'bg-blue-50 text-blue-700 border-blue-200',
    assigned: 'bg-indigo-50 text-indigo-700 border-indigo-200',
    consulted: 'bg-orange-50 text-orange-700 border-orange-200',
    completed: 'bg-green-50 text-green-700 border-green-200',
  };

  // Reusable classes
  const inputBaseClass = "w-full px-4 py-2.5 text-sm text-gray-900 bg-white border transition-colors duration-200 focus:outline-none focus:ring-1";
  const inputNormalClass = `${inputBaseClass} border-gray-300 focus:border-blue-600 focus:ring-blue-600`;
  const inputErrorClass = `${inputBaseClass} border-red-400 focus:border-red-600 focus:ring-red-600`;
  const labelClass = "block text-xs font-semibold uppercase tracking-wider text-gray-600 mb-1.5";

  return (
    <div className="flex min-h-screen bg-gray-50 font-sans antialiased">
      <Sidebar navItems={navItems} />
      <main className="flex-1 p-8 overflow-y-auto">
        <h1 className="text-2xl font-bold tracking-tight text-gray-900 border-b border-gray-200 pb-4 mb-6">
          {activeTab === 'analytics' && '📊 Analytics Dashboard'}
          {activeTab === 'doctors' && '👨‍⚕️ Manage Doctors'}
          {activeTab === 'receptionists' && '🧑‍💼 Manage Receptionists'}
          {activeTab === 'nextAppointments' && '📅 Next Appointments (Follow-ups)'}
          {activeTab === 'patients' && '👥 All Patients'}
        </h1>

        {/* ── Global Error Banner ── */}
        {error && (
          <div className="bg-red-50 border-l-4 border-red-600 text-red-800 px-4 py-3 mb-4 text-sm flex items-center justify-between shadow-sm">
            <div className="flex items-center space-x-2">
              <span className="font-bold">⚠️</span>
              <span>{error}</span>
            </div>
            <button onClick={() => setError('')} className="text-red-500 hover:text-red-800 ml-4 font-bold">✕</button>
          </div>
        )}

        {/* ── Analytics Tab ── */}
        {activeTab === 'analytics' && (
          <div className="space-y-6">
            {/* Date Range & Filters */}
            <div className="bg-white border border-gray-200 shadow-sm p-6">
              <div className="flex flex-wrap items-end gap-4">
                <div>
                  <label className={labelClass}>From Date</label>
                  <input
                    type="date"
                    value={fromDate}
                    onChange={(e) => setFromDate(e.target.value)}
                    className={`${inputNormalClass} w-auto`}
                  />
                </div>
                <div>
                  <label className={labelClass}>To Date</label>
                  <input
                    type="date"
                    value={toDate}
                    onChange={(e) => setToDate(e.target.value)}
                    className={`${inputNormalClass} w-auto`}
                  />
                </div>
                <div>
                  <label className={labelClass}>Doctor</label>
                  <select
                    value={analyticsDoctor}
                    onChange={(e) => setAnalyticsDoctor(e.target.value)}
                    className={`${inputNormalClass} w-auto`}
                  >
                    <option value="">All Doctors</option>
                    {doctors.map((doc) => (
                      <option key={doc.id} value={doc.id}>{doc.user?.name}</option>
                    ))}
                  </select>
                </div>
                <div>
                  <label className={labelClass}>Department</label>
                  <select
                    value={analyticsDept}
                    onChange={(e) => setAnalyticsDept(e.target.value)}
                    className={`${inputNormalClass} w-auto`}
                  >
                    <option value="">All Departments</option>
                    {analytics?.departments?.map((dept) => (
                      <option key={dept} value={dept}>{dept}</option>
                    ))}
                  </select>
                </div>
                <button
                  onClick={() => {
                    setFromDate(() => {
                      const d = new Date(); d.setDate(d.getDate() - 30);
                      return d.toISOString().split('T')[0];
                    });
                    setToDate(new Date().toISOString().split('T')[0]);
                    setAnalyticsDoctor('');
                    setAnalyticsDept('');
                  }}
                  className="px-4 py-2.5 text-xs font-bold uppercase tracking-wider bg-gray-100 text-gray-700 border border-gray-200 hover:bg-gray-200 transition-colors duration-200"
                >
                  Reset
                </button>
              </div>
            </div>

            {loading && !analytics ? (
              <div className="bg-white border border-gray-200 p-16 text-center shadow-sm">
                <Spinner size="lg" />
                <p className="mt-4 text-gray-500 font-medium uppercase tracking-wider text-xs">Loading analytics...</p>
              </div>
            ) : analytics ? (
              <>
                {/* Key Metrics */}
                <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                  <StatCard title="Total Patients" value={analytics.total_patients} color="blue" />
                  <StatCard title="Patients (Range)" value={analytics.range_patients} color="indigo" />
                  <StatCard title="Today's Patients" value={analytics.today_patients} color="emerald" />
                  <StatCard title="Active Doctors" value={analytics.active_doctors} color="purple" />
                </div>

                {/* Revenue */}
                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                  <StatCard title="Total Revenue (Range)" value={`₹${Number(analytics.total_revenue || 0).toLocaleString()}`} color="green" />
                  <StatCard title="Today's Revenue" value={`₹${Number(analytics.today_revenue || 0).toLocaleString()}`} color="emerald" />
                  <StatCard title="Monthly Revenue" value={`₹${Number(analytics.monthly_revenue || 0).toLocaleString()}`} color="teal" />
                </div>

                {/* Status Overview */}
                <div className="grid grid-cols-3 md:grid-cols-5 gap-4">
                  <StatCard title="Pending" value={analytics.assigned_patients} color="yellow" />
                  <StatCard title="In Consultation" value={analytics.consulted_patients} color="orange" />
                  <StatCard title="Completed" value={analytics.completed_patients} color="green" />
                  <StatCard title="Receptionists" value={analytics.total_receptionists} color="gray" />
                  <StatCard title="Total Doctors" value={analytics.total_doctors} color="blue" />
                </div>

                {/* Charts Row */}
                <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                  {/* Doctor-wise Patient Count */}
                  <div className="bg-white border border-gray-200 shadow-sm p-6">
                    <h3 className="text-lg font-bold tracking-tight text-gray-900 mb-6">👨‍⚕️ Patients per Doctor</h3>
                    <div className="space-y-4">
                      {analytics.doctor_wise_count?.length > 0 ? (
                        analytics.doctor_wise_count.map((item) => {
                          const maxCount = Math.max(...analytics.doctor_wise_count.map(d => d.total));
                          return (
                            <div key={item.doctor_id} className="flex items-center justify-between">
                              <span className="text-sm font-medium text-gray-700 w-32 truncate">{item.doctor?.user?.name || 'Unknown'}</span>
                              <div className="flex items-center space-x-3 flex-1 ml-4">
                                <div className="w-48 bg-gray-100 h-3">
                                  <div
                                    className="bg-blue-600 h-3 transition-all duration-500"
                                    style={{ width: `${maxCount > 0 ? (item.total / maxCount) * 100 : 0}%` }}
                                  />
                                </div>
                                <span className="text-sm font-bold text-gray-900 w-8 text-right">{item.total}</span>
                              </div>
                            </div>
                          );
                        })
                      ) : (
                        <p className="text-sm text-gray-500 font-medium">No data yet</p>
                      )}
                    </div>
                  </div>

                  {/* Doctor Revenue */}
                  <div className="bg-white border border-gray-200 shadow-sm p-6">
                    <h3 className="text-lg font-bold tracking-tight text-gray-900 mb-6">💰 Revenue per Doctor</h3>
                    <div className="space-y-4">
                      {analytics.doctor_revenue?.length > 0 ? (
                        analytics.doctor_revenue.map((item) => {
                          const maxRev = Math.max(...analytics.doctor_revenue.map(d => d.revenue));
                          return (
                            <div key={item.doctor_id} className="flex items-center justify-between">
                              <span className="text-sm font-medium text-gray-700 w-32 truncate">{item.name}</span>
                              <div className="flex items-center space-x-3 flex-1 ml-4">
                                <div className="w-48 bg-gray-100 h-3">
                                  <div
                                    className="bg-green-600 h-3 transition-all duration-500"
                                    style={{ width: `${maxRev > 0 ? (item.revenue / maxRev) * 100 : 0}%` }}
                                  />
                                </div>
                                <span className="text-sm font-bold text-gray-900 w-20 text-right">₹{item.revenue.toLocaleString()}</span>
                              </div>
                            </div>
                          );
                        })
                      ) : (
                        <p className="text-sm text-gray-500 font-medium">No revenue data</p>
                      )}
                    </div>
                  </div>
                </div>

                {/* Daily Stats & Status Distribution */}
                <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                  {/* Daily Patient Trend */}
                  <div className="bg-white border border-gray-200 shadow-sm p-6">
                    <h3 className="text-lg font-bold tracking-tight text-gray-900 mb-6">📈 Daily Patient Trend</h3>
                    {analytics.daily_stats?.length > 0 ? (
                      <div className="flex items-end gap-1.5 h-40">
                        {analytics.daily_stats.map((day) => {
                          const maxDay = Math.max(...analytics.daily_stats.map(d => d.total));
                          return (
                            <div key={day.date} className="flex-1 flex flex-col items-center gap-1 group relative">
                              <div
                                className="w-full bg-blue-600 transition-all duration-300 hover:bg-blue-700"
                                style={{ height: `${maxDay > 0 ? (day.total / maxDay) * 100 : 0}%`, minHeight: '4px' }}
                              />
                              <span className="text-[10px] text-gray-500 font-medium hidden sm:inline">
                                {new Date(day.date).toLocaleDateString('en', { month: 'short', day: 'numeric' })}
                              </span>
                              <div className="absolute -top-8 bg-gray-900 text-white text-xs px-2 py-1 hidden group-hover:block whitespace-nowrap shadow-lg">
                                {day.date}: {day.total} patients
                              </div>
                            </div>
                          );
                        })}
                      </div>
                    ) : (
                      <p className="text-sm text-gray-500 font-medium">No daily data</p>
                    )}
                  </div>

                  {/* Appointment Status Distribution */}
                  <div className="bg-white border border-gray-200 shadow-sm p-6">
                    <h3 className="text-lg font-bold tracking-tight text-gray-900 mb-6">📊 Appointment Status Distribution</h3>
                    {analytics.status_distribution?.length > 0 ? (
                      <div className="space-y-4">
                        {analytics.status_distribution.map((item) => {
                          const total = analytics.status_distribution.reduce((sum, s) => sum + s.count, 0);
                          const pct = total > 0 ? ((item.count / total) * 100).toFixed(1) : 0;
                          const barColors = {
                            pending: 'bg-yellow-500',
                            enquired: 'bg-blue-500',
                            assigned: 'bg-indigo-500',
                            consulted: 'bg-orange-500',
                            completed: 'bg-green-500',
                          };
                          return (
                            <div key={item.status} className="flex items-center space-x-3">
                              <span className="text-sm font-medium text-gray-700 w-24 capitalize">{item.status}</span>
                              <div className="flex-1 bg-gray-100 h-4">
                                <div
                                  className={`${barColors[item.status] || 'bg-gray-500'} h-4 transition-all duration-500`}
                                  style={{ width: `${pct}%` }}
                                />
                              </div>
                              <span className="text-sm font-bold text-gray-900 w-20 text-right">{item.count} ({pct}%)</span>
                            </div>
                          );
                        })}
                      </div>
                    ) : (
                      <p className="text-sm text-gray-500 font-medium">No status data</p>
                    )}
                  </div>
                </div>
              </>
            ) : (
              <div className="bg-white border border-gray-200 shadow-sm p-16 text-center">
                <p className="text-gray-500 font-medium">No analytics data available</p>
              </div>
            )}
          </div>
        )}

        {/* ── Next Appointments Tab ── */}
        {activeTab === 'nextAppointments' && (
          <div className="space-y-4">
            <div className="bg-blue-50 border-l-4 border-blue-600 p-4 text-sm text-blue-800 shadow-sm">
              ℹ️ Follow-up appointments set by doctors during consultation. Use the date filter to view patients for a specific day.
            </div>

            {/* Date & Doctor filter */}
            <div className="flex items-center gap-4 bg-white border border-gray-200 shadow-sm p-4 flex-wrap">
              <div className="flex items-center gap-2">
                <label className="text-xs font-bold uppercase tracking-wider text-gray-600">📅 Date:</label>
                <input
                  type="date"
                  value={nextApptDate}
                  onChange={(e) => setNextApptDate(e.target.value)}
                  className={`${inputNormalClass} w-auto`}
                />
                <button
                  onClick={() => setNextApptDate('')}
                  className="px-3 py-2.5 text-xs font-bold uppercase tracking-wider bg-gray-100 text-gray-700 border border-gray-200 hover:bg-gray-200 transition-colors duration-200"
                >
                  All Upcoming
                </button>
              </div>

              <div className="flex items-center gap-2 ml-auto">
                <label className="text-xs font-bold uppercase tracking-wider text-gray-600">👨‍⚕️ Doctor:</label>
                <select
                  value={nextApptDoctor}
                  onChange={(e) => setNextApptDoctor(e.target.value)}
                  className={`${inputNormalClass} w-auto`}
                >
                  <option value="">All Doctors</option>
                  {doctors.map((doc) => (
                    <option key={doc.id} value={doc.id}>{doc.user?.name}</option>
                  ))}
                </select>
                <button
                  onClick={() => { setNextApptDate(''); setNextApptDoctor(''); }}
                  className="px-3 py-2.5 text-xs font-bold uppercase tracking-wider bg-blue-50 text-blue-700 border border-blue-200 hover:bg-blue-100 transition-colors duration-200"
                >
                  Reset Filters
                </button>
              </div>
            </div>

            {/* Summary cards */}
            <div className="grid grid-cols-3 gap-4">
              <div className="bg-white border border-gray-200 p-5 shadow-sm">

        const params = new URLSearchParams();
        if (followupDate) params.append('date', followupDate);
        const res = await api.get(`/doctor/today-followups?${params.toString()}`);
        setFollowups(res.data);
      }
    } catch (err) {
      console.error('[Doctor] Fetch failed:', getErrorMessage(err));
    }
  }, [activeTab, followupDate]);

  // Auto-refresh polling (fallback)
  useEffect(() => {
    fetchData();
    pollRef.current = setInterval(fetchData, POLL_INTERVAL);
    return () => clearInterval(pollRef.current);
  }, [fetchData]);

  // ── Real-time Echo subscriptions ──
  const { user } = useAuth();
  const doctorId = user?.doctor?.id;

  useEchoChannel(
    `doctor.${doctorId}`,
    '.patient.created',
    useCallback((e) => {
      console.log('[Echo] New patient assigned:', e);
      fetchData();
      showMsg('success', `📩 ${e.message}`);
    }, [fetchData])
  );

  useEchoChannel(
    `doctor.${doctorId}`,
    '.patient.status.updated',
    useCallback((e) => {
      console.log('[Echo] Patient status updated:', e);
      fetchData();
    }, [fetchData])
  );

  useEchoChannel(
    `doctor.${doctorId}`,
    '.consultation.updated',
    useCallback((e) => {
      console.log('[Echo] Consultation updated:', e);
      fetchData();
    }, [fetchData])
  );

  const showMsg = (type, text) => {
    setMsg({ type, text });
    setTimeout(() => setMsg({ type: '', text: '' }), 3000);
  };

  const startConsultation = async (patientId) => {
    try {
      await api.post(`/doctor/start-consultation/${patientId}`);
      showMsg('success', 'Consultation started!');
      fetchData();
      setActiveTab('consulted');
    } catch (err) {
      showMsg('error', getErrorMessage(err));
    }
  };

  const openConsultation = (patient) => {
    setSelectedPatient(patient);
    setConsultationForm({
      diagnosis: patient.consultation?.diagnosis || '',
      treatment: patient.consultation?.treatment || '',
      prescription: patient.consultation?.prescription || '',
      notes: patient.consultation?.notes || '',
      next_appointment: patient.consultation?.next_appointment
        ? (typeof patient.consultation.next_appointment === 'string'
            ? patient.consultation.next_appointment.split('T')[0]
            : new Date(patient.consultation.next_appointment).toISOString().split('T')[0])
        : '',
      temperature: patient.consultation?.temperature || '',
      blood_pressure: patient.consultation?.blood_pressure || '',
      heart_rate: patient.consultation?.heart_rate || '',
    });
  };

  const saveConsultation = async () => {
    setSaving(true);
    try {
      const consultationId = selectedPatient.consultation?.id;
      if (consultationId) {
        await api.put(`/doctor/consultation/${consultationId}`, consultationForm);
        showMsg('success', 'Consultation saved!');
        setSelectedPatient(null);
        fetchData();
      }
    } catch (err) {
      showMsg('error', getErrorMessage(err));
    } finally {
      setSaving(false);
    }
  };

  const completeConsultation = async () => {
    setSaving(true);
    try {
      const consultationId = selectedPatient.consultation?.id;
      if (consultationId) {
        await api.post(`/doctor/complete-consultation/${consultationId}`, consultationForm);
        showMsg('success', consultationForm.next_appointment
          ? 'Consultation completed! Follow-up scheduled.'
          : 'Consultation completed!');
        setSelectedPatient(null);
        fetchData();
        setActiveTab('completed');
      }
    } catch (err) {
      showMsg('error', getErrorMessage(err));
    } finally {
      setSaving(false);
    }
  };

  const loadVisitHistory = async (patientId) => {
    setLoadingHistory(true);
    try {
      const res = await api.get(`/doctor/visit-history/${patientId}`);
      setVisitHistoryPatient(res.data.patient);
      setVisitHistory(res.data.visits || []);
    } catch (err) {
      showMsg('error', getErrorMessage(err));
    } finally {
      setLoadingHistory(false);
    }
  };

  const sendWhatsAppReminder = async (patientId) => {
    try {
      const res = await api.post(`/doctor/whatsapp-reminder/${patientId}`);
      window.open(res.data.whatsapp_url, '_blank');
      fetchData();
    } catch (err) {
      showMsg('error', getErrorMessage(err));
    }
  };

  const handleWhatsAppEnquiry = (patient) => {
    const hospitalName = 'HealthCare';
    const doctorName = 'You';
    const appointmentDate = patient.consultation?.next_appointment
      ? new Date(patient.consultation.next_appointment).toLocaleDateString('en-IN')
      : new Date().toLocaleDateString('en-IN');
    const appointmentTime = patient.consultation?.appointment_time
      ? new Date(patient.consultation.appointment_time).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
      : 'your scheduled time';

    const message = `Hello, this is regarding your appointment at ${hospitalName}.\n\nPatient: ${patient.name}\nDoctor: ${doctorName}\nScheduled Date: ${appointmentDate}\nTime: ${appointmentTime}\n\nPlease confirm your availability.`;

    const url = generateWhatsAppLink(patient.phone, message);
    window.open(url, '_blank');
  };

  const navItems = [
    { id: 'enquiries', label: 'New Enquiries', icon: '📩', active: activeTab === 'enquiries', onClick: () => setActiveTab('enquiries'), badge: enquiries.length },
    { id: 'consulted', label: 'In Consultation', icon: '🩺', active: activeTab === 'consulted', onClick: () => setActiveTab('consulted') },
    { id: 'completed', label: 'Completed', icon: '✅', active: activeTab === 'completed', onClick: () => setActiveTab('completed') },
    { id: 'followups', label: "Today's Follow-ups", icon: '📅', active: activeTab === 'followups', onClick: () => setActiveTab('followups'), badge: followups.length },
  ];

  const VisitBadge = ({ status }) => {
    const config = STATUS_CONFIG[status] || STATUS_CONFIG.pending;
    return (
      <span className={`px-2 py-1 text-xs font-semibold uppercase tracking-wider border ${config.color}`}>
        {config.dot} {config.label}
      </span>
    );
  };

  // Reusable classes
  const inputBaseClass = "w-full px-4 py-2.5 text-sm text-gray-900 bg-white border transition-colors duration-200 focus:outline-none focus:ring-1";
  const inputNormalClass = `${inputBaseClass} border-gray-300 focus:border-blue-600 focus:ring-blue-600`;
  const labelClass = "block text-xs font-semibold uppercase tracking-wider text-gray-600 mb-1.5";

  // Patient card component (shared)
  const PatientCard = ({ patient, actions, accentColor = 'blue' }) => {
    const accent = ACCENT_MAP[accentColor] || ACCENT_MAP.blue;
    return (
      <div className="bg-white border border-gray-200 shadow-sm p-6 transition-all duration-200 hover:border-blue-400 hover:shadow-md">
        <div className="flex items-center justify-between gap-4">
          <div className="flex items-center space-x-4">
            <div className={`w-12 h-12 border flex items-center justify-center font-bold uppercase tracking-wider ${accent.box}`}>
              {patient.name.charAt(0)}
            </div>
            <div>
              <div className="flex items-center gap-2">
                <h3 className="font-bold text-gray-900 tracking-tight">{patient.name}</h3>
                {patient.relation && patient.relation !== 'Myself' && (
                  <span className="px-2 py-0.5 text-xs font-semibold uppercase tracking-wider bg-gray-100 text-gray-600 border border-gray-200">{patient.relation}</span>
                )}
              </div>
              <p className="text-sm text-gray-500 font-medium">{patient.age} yrs • {patient.gender} • {patient.phone}</p>
              {patient.consultation?.symptoms && (
                <p className="text-sm text-gray-700 mt-1 bg-gray-50 px-2 py-1 border border-gray-100">Symptoms: {patient.consultation.symptoms}</p>
              )}
              {(patient.consultation?.temperature || patient.consultation?.blood_pressure || patient.consultation?.heart_rate) && (
                <div className="flex gap-3 mt-1 text-xs text-gray-600 font-medium">
                  {patient.consultation.temperature && <span>🌡️ {patient.consultation.temperature}</span>}
                  {patient.consultation.blood_pressure && <span>🩸 {patient.consultation.blood_pressure}</span>}
                  {patient.consultation.heart_rate && <span>💓 {patient.consultation.heart_rate}</span>}
                </div>
              )}
              <p className="text-xs text-gray-400 mt-1 uppercase tracking-wider">Assigned by: {patient.creator?.name}</p>
            </div>
          </div>
          <div className="flex-shrink-0">{actions}</div>
        </div>
      </div>
    );
  };

  return (
    <div className="flex min-h-screen bg-gray-50 font-sans antialiased">
      <Sidebar navItems={navItems} />
      <main className="flex-1 p-8 overflow-y-auto">
        <h1 className="text-2xl font-bold tracking-tight text-gray-900 border-b border-gray-200 pb-4 mb-6">
          {activeTab === 'enquiries' && '📩 New Enquiries'}
          {activeTab === 'consulted' && '🩺 Patients In Consultation'}
          {activeTab === 'completed' && '✅ Completed Cases'}
          {activeTab === 'followups' && "📅 Today's Follow-ups"}
        </h1>

        {msg.text && (
          <div className={`px-4 py-3 mb-4 text-sm flex items-center space-x-2 shadow-sm ${
            msg.type === 'success'
              ? 'bg-green-50 border-l-4 border-green-600 text-green-800'
              : 'bg-red-50 border-l-4 border-red-600 text-red-800'
          }`}>
            <span className="font-bold">{msg.type === 'success' ? '✅' : '⚠️'}</span>
            <span>{msg.text}</span>
          </div>
        )}

        {/* ── New Enquiries ── */}
        {activeTab === 'enquiries' && (
          <div className="space-y-4">
            {enquiries.length === 0 ? (
              <EmptyState icon="📭" message="No new enquiries" />
            ) : (
              enquiries.map((patient) => (
                <PatientCard
                  key={patient.id}
                  patient={patient}
                  accentColor="blue"
                  actions={
                    <div className="flex gap-2">
                      <button
                        onClick={() => loadVisitHistory(patient.id)}
                        className="px-3 py-2 bg-gray-100 text-gray-700 text-xs font-bold uppercase tracking-wider hover:bg-gray-200 transition-colors duration-200 border border-gray-200"
                        title="View visit history"
                      >
                        📋 History
                      </button>
                      <button
                        onClick={() => startConsultation(patient.id)}
                        className="px-4 py-2 bg-blue-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-blue-700 transition-colors duration-200 shadow-sm"
                      >
                        Start Consult
                      </button>
                    </div>
                  }
                />
              ))
            )}
          </div>
        )}

        {/* ── In Consultation ── */}
        {activeTab === 'consulted' && (
          <div className="space-y-4">
            {consulted.length === 0 ? (
              <EmptyState icon="🩺" message="No patients in consultation" />
            ) : (
              consulted.map((patient) => (
                <PatientCard
                  key={patient.id}
                  patient={patient}
                  accentColor="orange"
                  actions={
                    <div className="flex gap-2 items-center">
                      <button
                        onClick={() => loadVisitHistory(patient.id)}
                        className="px-3 py-2 bg-gray-100 text-gray-700 text-xs font-bold uppercase tracking-wider hover:bg-gray-200 transition-colors duration-200 border border-gray-200"
                        title="View visit history"
                      >
                        📋 History
                      </button>
                      {patient.consultation?.diagnosis && (
                        <p className="text-xs text-blue-700 font-semibold uppercase tracking-wider">📝 In Progress</p>
                      )}
                      <button
                        onClick={() => openConsultation(patient)}
                        className="px-4 py-2 bg-green-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-green-700 transition-colors duration-200 shadow-sm"
                      >
                        📝 Open
                      </button>
                    </div>
                  }
                />
              ))
            )}
          </div>
        )}

        {/* ── Completed Cases ── */}
        {activeTab === 'completed' && (
          <div className="space-y-4">
            {completed.length === 0 ? (
              <EmptyState icon="✅" message="No completed cases" />
            ) : (
              completed.map((patient) => (
                <PatientCard
                  key={patient.id}
                  patient={patient}
                  accentColor="green"
                  actions={
                    <div className="flex items-center gap-3 flex-wrap justify-end">
                      <button
                        onClick={() => loadVisitHistory(patient.id)}
                        className="px-3 py-2 bg-gray-100 text-gray-700 text-xs font-bold uppercase tracking-wider hover:bg-gray-200 transition-colors duration-200 border border-gray-200"
                      >
                        📋 History
                      </button>
                      {patient.consultation?.diagnosis && (
                        <p className="text-xs text-gray-600 bg-gray-50 px-2 py-1 border border-gray-200 max-w-[150px] truncate" title={patient.consultation.diagnosis}>
                          {patient.consultation.diagnosis}
                        </p>
                      )}
                      {patient.consultation?.next_appointment && (
                        <p className="text-xs text-blue-700 font-bold uppercase tracking-wider">
                          📅 {new Date(patient.consultation.next_appointment).toLocaleDateString()}
                        </p>
                      )}
                      <span className="px-3 py-1 bg-green-50 text-green-700 border border-green-200 text-xs font-bold uppercase tracking-wider">✅ Done</span>
                    </div>
                  }
                />
              ))
            )}
          </div>
        )}

        {/* ── Today's Follow-ups ── */}
        {activeTab === 'followups' && (
          <div className="space-y-4">
            <div className="flex items-center gap-3 bg-white p-4 border border-gray-200 shadow-sm flex-wrap">
              <label className="text-xs font-bold uppercase tracking-wider text-gray-600">📅 Follow-up Date:</label>
              <input
                type="date"
                value={followupDate}
                onChange={(e) => setFollowupDate(e.target.value)}
                className={`${inputNormalClass} w-auto`}
              />
              <button
                onClick={() => setFollowupDate(new Date().toISOString().split('T')[0])}
                className="px-4 py-2 text-xs font-bold uppercase tracking-wider bg-blue-50 text-blue-700 border border-blue-200 hover:bg-blue-100 transition-colors duration-200"
              >
                Today
              </button>
            </div>

            {followups.length === 0 ? (
              <EmptyState icon="📅" message={`No follow-ups scheduled for ${new Date(followupDate + 'T00:00:00').toLocaleDateString()}`} />
            ) : (
              followups.map((patient) => (
                <PatientCard
                  key={patient.id}
                  patient={patient}
                  accentColor="purple"
                  actions={
                    <div className="flex gap-2 items-center flex-wrap justify-end">
                      {patient.consultation?.next_appointment && (
                        <p className="text-xs text-blue-700 font-bold uppercase tracking-wider mr-2">
                          📅 {new Date(patient.consultation.next_appointment).toLocaleDateString()}
                        </p>
                      )}
                      <button
                        onClick={() => openConsultation(patient)}
                        className="px-4 py-2 bg-green-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-green-700 transition-colors duration-200 shadow-sm"
                      >
                        📝 Consult
                      </button>
                      <button
                        onClick={() => sendWhatsAppReminder(patient.id)}
                        className="flex items-center space-x-1 px-4 py-2 bg-gray-100 text-gray-700 text-xs font-bold uppercase tracking-wider hover:bg-gray-200 transition-colors duration-200 border border-gray-200"
                      >
                        <span>📱</span>
                        <span>App</span>
                      </button>
                      <button
                        onClick={() => handleWhatsAppEnquiry(patient)}
                        className="flex items-center space-x-1 px-4 py-2 bg-blue-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-blue-700 transition-colors duration-200 shadow-sm"
                        title="Send WhatsApp enquiry with appointment details"
                      >
                        <span>💬</span>
                        <span>Enquire</span>
                      </button>
                    </div>
                  }
                />
              ))
            )}
          </div>
        )}

        {/* ── Consultation Modal ── */}
        {selectedPatient && (
          <div className="fixed inset-0 bg-black/60 backdrop-blur-sm flex items-center justify-center z-50 p-4">
            <div className="bg-white border border-gray-200 p-8 w-full max-w-2xl shadow-xl max-h-[90vh] overflow-y-auto">
              <div className="flex items-center justify-between mb-6 border-b border-gray-200 pb-4">
                <h3 className="text-xl font-bold tracking-tight text-gray-900">
                  Consultation — <span className="text-blue-600">{selectedPatient.name}</span>
                </h3>
                <button onClick={() => setSelectedPatient(null)} className="text-gray-400 hover:text-gray-900 text-3xl leading-none">&times;</button>
              </div>

              <div className="space-y-5">
                {(selectedPatient.consultation?.temperature || selectedPatient.consultation?.blood_pressure || selectedPatient.consultation?.heart_rate) && (
                  <div className="bg-blue-50 border-l-4 border-blue-600 p-4">
                    <p className="text-xs font-bold uppercase tracking-wider text-blue-800 mb-2">🩺 Visit Vitals (Recorded at Registration)</p>
                    <div className="grid grid-cols-3 gap-3 text-sm">
                      {selectedPatient.consultation.temperature && (
                        <div><span className="text-gray-500 text-xs uppercase tracking-wider">🌡️ Temp:</span> <span className="font-bold text-gray-900">{selectedPatient.consultation.temperature}</span></div>
                      )}
                      {selectedPatient.consultation.blood_pressure && (
                        <div><span className="text-gray-500 text-xs uppercase tracking-wider">🩸 BP:</span> <span className="font-bold text-gray-900">{selectedPatient.consultation.blood_pressure}</span></div>
                      )}
                      {selectedPatient.consultation.heart_rate && (
                        <div><span className="text-gray-500 text-xs uppercase tracking-wider">💓 HR:</span> <span className="font-bold text-gray-900">{selectedPatient.consultation.heart_rate}</span></div>
                      )}
                    </div>
                  </div>
                )}

                {selectedPatient.consultation?.symptoms && (
                  <div className="bg-yellow-50 border-l-4 border-yellow-500 p-4">
                    <p className="text-xs font-bold uppercase tracking-wider text-yellow-800 mb-1">📋 Visit Symptoms</p>
                    <p className="text-sm text-yellow-900 font-medium">{selectedPatient.consultation.symptoms}</p>
                  </div>
                )}

                <div className="grid grid-cols-3 gap-4">
                  <div>
                    <label className={labelClass}>🌡️ Temperature</label>
                    <input
                      type="text"
                      value={consultationForm.temperature}
                      onChange={(e) => setConsultationForm({ ...consultationForm, temperature: e.target.value })}
                      className={inputNormalClass}
                      placeholder="e.g. 98.6°F"
                    />
                  </div>
                  <div>
                    <label className={labelClass}>🩸 Blood Pressure</label>
                    <input
                      type="text"
                      value={consultationForm.blood_pressure}
                      onChange={(e) => setConsultationForm({ ...consultationForm, blood_pressure: e.target.value })}
                      className={inputNormalClass}
                      placeholder="e.g. 120/80"
                    />
                  </div>
                  <div>
                    <label className={labelClass}>💓 Heart Rate</label>
                    <input
                      type="text"
                      value={consultationForm.heart_rate}
                      onChange={(e) => setConsultationForm({ ...consultationForm, heart_rate: e.target.value })}
                      className={inputNormalClass}
                      placeholder="e.g. 72 bpm"
                    />
                  </div>
                </div>

                <div>
                  <label className={labelClass}>Diagnosis</label>
                  <textarea
                    value={consultationForm.diagnosis}
                    onChange={(e) => setConsultationForm({ ...consultationForm, diagnosis: e.target.value })}
                    className={inputNormalClass}
                    rows={3}
                    placeholder="Enter diagnosis..."
                  />
                </div>
                <div>
                  <label className={labelClass}>Treatment</label>
                  <textarea
                    value={consultationForm.treatment}
                    onChange={(e) => setConsultationForm({ ...consultationForm, treatment: e.target.value })}
                    className={inputNormalClass}
                    rows={3}
                    placeholder="Enter treatment plan..."
                  />
                </div>
                <div>
                  <label className={labelClass}>Prescription</label>
                  <textarea
                    value={consultationForm.prescription}
                    onChange={(e) => setConsultationForm({ ...consultationForm, prescription: e.target.value })}
                    className={inputNormalClass}
                    rows={3}
                    placeholder="Enter prescription..."
                  />
                </div>
                <div>
                  <label className={labelClass}>Notes</label>
                  <textarea
                    value={consultationForm.notes}
                    onChange={(e) => setConsultationForm({ ...consultationForm, notes: e.target.value })}
                    className={inputNormalClass}
                    rows={2}
                    placeholder="Additional notes..."
                  />
                </div>
                <div>
                  <label className={labelClass}>
                    📅 Next Appointment
                    <span className="text-gray-400 font-normal normal-case tracking-normal ml-2">(Optional — leave empty if no follow-up needed)</span>
                  </label>
                  <input
                    type="date"
                    value={consultationForm.next_appointment}
                    onChange={(e) => setConsultationForm({ ...consultationForm, next_appointment: e.target.value })}
                    className={inputNormalClass}
                  />
                  {consultationForm.next_appointment && (
                    <p className="mt-2 text-xs text-blue-700 font-bold uppercase tracking-wider">
                      ✅ Follow-up scheduled for {new Date(consultationForm.next_appointment).toLocaleDateString()}
                    </p>
                  )}
                </div>
              </div>

              <div className="flex justify-end space-x-3 mt-8 pt-5 border-t border-gray-200">
                <button
                  onClick={() => setSelectedPatient(null)}
                  disabled={saving}
                  className="px-5 py-2.5 text-sm font-bold uppercase tracking-wider text-gray-600 hover:text-gray-900 disabled:opacity-50 transition-colors"
                >
                  Cancel
                </button>
                <button
                  onClick={saveConsultation}
                  disabled={saving}
                  className="px-6 py-2.5 bg-blue-600 text-white text-sm font-bold uppercase tracking-wider hover:bg-blue-700 disabled:opacity-60 flex items-center space-x-2 shadow-sm transition-colors"
                >
                  {saving && <Spinner />}
                  <span>{saving ? 'Saving...' : 'Save Progress'}</span>
                </button>
                <button
                  onClick={completeConsultation}
                  disabled={saving}
                  className="px-6 py-2.5 bg-green-600 text-white text-sm font-bold uppercase tracking-wider hover:bg-green-700 disabled:opacity-60 flex items-center space-x-2 shadow-sm transition-colors"
                >
                  {saving && <Spinner />}
                  <span>{saving ? 'Saving...' : 'Complete & Save'}</span>
                </button>
              </div>
            </div>
          </div>
        )}

        {/* ── Visit History Modal ── */}
        {visitHistoryPatient && (
          <div className="fixed inset-0 bg-black/60 backdrop-blur-sm flex items-center justify-center z-50 p-4">
            <div className="bg-white border border-gray-200 p-8 w-full max-w-3xl shadow-xl max-h-[90vh] overflow-y-auto">
              <div className="flex items-center justify-between mb-6 border-b border-gray-200 pb-4">
                <h3 className="text-xl font-bold tracking-tight text-gray-900">
                  📋 Visit History — <span className="text-blue-600">{visitHistoryPatient.name}</span>
                </h3>
                <button
                  onClick={() => { setVisitHistoryPatient(null); setVisitHistory([]); }}
                  className="text-gray-400 hover:text-gray-900 text-3xl leading-none"
                >
                  &times;
                </button>
              </div>

              <div className="mb-6 bg-gray-50 border border-gray-200 p-4">
                <div className="grid grid-cols-2 md:grid-cols-4 gap-4 text-sm">
                  <div>
                    <span className="text-gray-500 text-xs uppercase tracking-wider block">Patient</span>
                    <strong className="text-gray-900 font-bold">{visitHistoryPatient.name}</strong>
                  </div>
                  <div>
                    <span className="text-gray-500 text-xs uppercase tracking-wider block">Phone</span>
                    <strong className="text-gray-900 font-bold">{visitHistoryPatient.phone}</strong>
                  </div>
                  <div>
                    <span className="text-gray-500 text-xs uppercase tracking-wider block">Age</span>
                    <strong className="text-gray-900 font-bold">{visitHistoryPatient.age} yrs</strong>
                  </div>
                  <div>
                    <span className="text-gray-500 text-xs uppercase tracking-wider block">Relation</span>
                    <strong className="text-gray-900 font-bold">{visitHistoryPatient.relation || 'Myself'}</strong>
                  </div>
                </div>
              </div>

              {loadingHistory ? (

                    <Spinner />
                  </div>
                )}
              </div>

              {phoneLookup && !lookupLoading && (
                <div className="mt-4">
                  {phoneLookup.exists ? (
                    <div className="bg-blue-50 border border-blue-200 p-4">
                      <div className="flex items-center justify-between mb-3">
                        <p className="text-sm font-medium text-blue-900">
                          ✅ Existing patient found: <span className="font-bold tracking-wide">{selectedPatient?.name}</span>
                          <span className="ml-2 text-xs bg-blue-200 text-blue-800 px-2 py-0.5 font-semibold uppercase tracking-wider">
                            {selectedPatient?.relation || 'Myself'}
                          </span>
                        </p>
                        <button
                          onClick={clearPhoneLookup}
                          className="text-xs text-blue-700 hover:text-blue-900 font-medium underline underline-offset-2"
                        >
                          ✕ Clear
                        </button>
                      </div>

                      {/* Existing profiles */}
                      {familyMembers.length > 0 && (
                        <div className="mb-4">
                          <label className="block text-xs font-semibold uppercase tracking-wider text-gray-700 mb-2">
                            👨‍👩‍👧 Select Profile:
                          </label>
                          <select
                            value={selectedPatient?.id || ''}
                            onChange={(e) => {
                              const member = familyMembers.find((m) => m.id === Number(e.target.value));
                              if (member) selectFamilyMember(member);
                            }}
                            className={`${inputNormalClass} mb-3`}
                          >
                            {familyMembers.map((member) => (
                              <option key={member.id} value={member.id}>
                                {member.name} ({member.relation || 'Myself'}) - {member.age} yrs
                              </option>
                            ))}
                          </select>
                        </div>
                      )}

                      {familyMembers.length > 0 && (
                        <div className="grid grid-cols-2 md:grid-cols-3 gap-3 mb-4">
                          {familyMembers.map((member) => (
                            <button
                              key={member.id}
                              onClick={() => selectFamilyMember(member)}
                              className={`p-3 border text-left transition-all duration-200 ${
                                selectedPatient?.id === member.id
                                  ? 'border-blue-600 bg-white ring-1 ring-blue-600 shadow-sm'
                                  : 'border-gray-200 bg-white hover:border-blue-400 hover:bg-blue-50'
                              }`}
                            >
                              <p className="text-sm font-bold text-gray-900">{member.name}</p>
                              <p className="text-xs text-gray-600">{member.relation || 'Myself'} • {member.age} yrs</p>
                              <p className="text-xs text-gray-400 mt-1">{member.phone}</p>
                            </button>
                          ))}
                        </div>
                      )}

                      {/* Visit History (filtered by selected member) */}
                      <div className="mt-3">
                        <p className="text-xs font-semibold uppercase tracking-wider text-gray-700 mb-2">
                          📋 Visit History for <strong>{selectedPatient?.name}</strong> ({visitHistory.length} visit{visitHistory.length !== 1 ? 's' : ''}):
                        </p>
                        {visitHistory.length > 0 ? (
                          <div className="space-y-2 max-h-48 overflow-y-auto pr-1">
                            {visitHistory.slice(0, 5).map((visit) => (
                              <div key={visit.id} className="flex items-center justify-between bg-white border border-gray-100 px-3 py-2">
                                <div>
                                  <p className="text-xs font-bold text-gray-800">Dr. {visit.doctor?.user?.name || 'Unknown'}</p>
                                  <p className="text-xs text-gray-500">{visit.symptoms || 'No symptoms'}</p>
                                  {visit.diagnosis && (
                                    <p className="text-xs text-blue-700 mt-0.5 font-medium">Diagnosis: {visit.diagnosis}</p>
                                  )}
                                </div>
                                <div className="text-right">
                                  <StatusBadge status={visit.status} />
                                  <p className="text-xs text-gray-400 mt-1">
                                    {new Date(visit.created_at).toLocaleDateString()}
                                  </p>
                                </div>
                              </div>
                            ))}
                          </div>
                        ) : (
                          <p className="text-xs text-gray-400 italic">No previous visits found for {selectedPatient?.name}.</p>
                        )}
                      </div>

                      {/* Action mode toggle */}
                      <div className="mt-4 flex items-center gap-3">
                        <button
                          type="button"
                          onClick={() => switchToNewVisit(null)}
                          className={`flex-1 px-4 py-3 text-sm font-semibold uppercase tracking-wider transition-all duration-200 border-2 ${
                            formMode === 'visit'
                              ? 'border-green-600 bg-green-50 text-green-700'
                              : 'border-gray-200 bg-white text-gray-500 hover:border-green-400 hover:text-green-600'
                          }`}
                        >
                          🔄 New Visit for {selectedPatient?.name}
                        </button>
                        <button
                          type="button"
                          onClick={switchToAddMember}
                          className={`flex-1 px-4 py-3 text-sm font-semibold uppercase tracking-wider transition-all duration-200 border-2 ${
                            formMode === 'addMember'
                              ? 'border-blue-600 bg-blue-50 text-blue-700'
                              : 'border-gray-200 bg-white text-gray-500 hover:border-blue-400 hover:text-blue-600'
                          }`}
                        >
                          ➕ Add Family Member
                        </button>
                      </div>

                      {formMode === 'visit' && (
                        <div className="mt-3 bg-green-50 border-l-4 border-green-600 p-3">
                          <p className="text-sm text-green-800 font-bold">
                            🔄 Ready to create a new visit for {selectedPatient?.name}
                          </p>
                          <p className="text-xs text-green-700 mt-1">
                            Fill in the visit details below. The patient profile will remain unchanged.
                          </p>
                        </div>
                      )}
                      {formMode === 'addMember' && (
                        <div className="mt-3 bg-blue-50 border-l-4 border-blue-600 p-3">
                          <p className="text-sm text-blue-800 font-bold">
                            ➕ Adding a new family member under {phoneLookup?.patient?.phone}
                          </p>
                          <p className="text-xs text-blue-700 mt-1">
                            Fill in the new member's details below. They will be linked to this family.
                          </p>
                        </div>
                      )}
                    </div>
                  ) : (
                    <div className="bg-yellow-50 border-l-4 border-yellow-500 p-4">
                      <p className="text-sm text-yellow-800 font-medium">
                        🆕 New patient — please fill the registration form below
                      </p>
                    </div>
                  )}
                </div>
              )}
            </div>

            {/* Registration / New Visit Form */}
            <form onSubmit={handleSubmit} className="bg-white border border-gray-200 shadow-sm p-6 space-y-5">
              <h2 className="text-lg font-bold text-gray-900 border-b border-gray-100 pb-3">
                {isReturningPatient ? (
                  <span className="flex items-center gap-2">
                    🔄 New Visit — <span className="text-blue-600">{selectedPatient.name}</span>
                  </span>
                ) : isAddingMember ? (
                  <span className="flex items-center gap-2">
                    ➕ Add Family Member — <span className="text-blue-600">{phoneLookup?.patient?.phone}</span>
                  </span>
                ) : '🆕 New Patient Registration'}
              </h2>

              {/* Profile fields (shown for new patients, read-only for returning) */}
              {isReturningPatient ? (
                <div className="bg-gray-50 border border-gray-200 p-4">
                  <p className="text-xs font-semibold uppercase tracking-wider text-gray-500 mb-3">Patient Profile (Existing Record)</p>
                  <div className="grid grid-cols-2 gap-4 text-sm">
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Name:</span>
                      <p className="font-medium text-gray-900">{selectedPatient.name}</p>
                    </div>
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Age:</span>
                      <p className="font-medium text-gray-900">{selectedPatient.age} yrs</p>
                    </div>
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Gender:</span>
                      <p className="font-medium text-gray-900 capitalize">{selectedPatient.gender}</p>
                    </div>
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Phone:</span>
                      <p className="font-medium text-gray-900">{selectedPatient.phone}</p>
                    </div>
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Relation:</span>
                      <p className="font-medium text-gray-900">{selectedPatient.relation || 'Myself'}</p>
                    </div>
                    <div>
                      <span className="text-gray-500 text-xs uppercase tracking-wider">Total Visits:</span>
                      <p className="font-medium text-gray-900">{visitHistory.length}</p>
                    </div>
                  </div>
                </div>
              ) : (
                <>
                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className={labelClass}>Patient Name *</label>
                      <input
                        type="text"
                        value={formData.name}
                        onChange={(e) => {
                          setFormData({ ...formData, name: e.target.value });
                          if (touched.name) setFieldErrors((p) => ({ ...p, name: validateField('name', e.target.value) }));
                        }}
                        onBlur={() => handleBlur('name')}
                        className={touched.name && fieldErrors.name ? inputErrorClass : inputNormalClass}
                        placeholder="Enter full name"
                      />
                      {touched.name && fieldErrors.name && (
                        <p className="mt-1 text-xs text-red-600 font-medium">• {fieldErrors.name}</p>
                      )}
                    </div>
                    <div>
                      <label className={labelClass}>Age *</label>
                      <input
                        type="number"
                        value={formData.age}
                        onChange={(e) => {
                          setFormData({ ...formData, age: e.target.value });
                          if (touched.age) setFieldErrors((p) => ({ ...p, age: validateField('age', e.target.value) }));
                        }}
                        onBlur={() => handleBlur('age')}
                        className={touched.age && fieldErrors.age ? inputErrorClass : inputNormalClass}
                        placeholder="Age"
                        min="1"
                        max="150"
                      />
                      {touched.age && fieldErrors.age && (
                        <p className="mt-1 text-xs text-red-600 font-medium">• {fieldErrors.age}</p>
                      )}
                    </div>
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className={labelClass}>Gender *</label>
                      <select
                        value={formData.gender}
                        onChange={(e) => setFormData({ ...formData, gender: e.target.value })}
                        className={inputNormalClass}
                      >
                        <option value="male">Male</option>
                        <option value="female">Female</option>
                        <option value="other">Other</option>
                      </select>
                    </div>
                    <div>
                      <label className={labelClass}>Phone Number *</label>
                      <input
                        type="text"
                        value={formData.phone}
                        onChange={(e) => {
                          setFormData({ ...formData, phone: e.target.value });
                          if (touched.phone) setFieldErrors((p) => ({ ...p, phone: validateField('phone', e.target.value) }));
                        }}
                        onBlur={() => handleBlur('phone')}
                        readOnly={isAddingMember}
                        className={`${isAddingMember ? 'bg-gray-100 text-gray-500 cursor-not-allowed border-gray-200' : touched.phone && fieldErrors.phone ? inputErrorClass : inputNormalClass}`}
                        placeholder="Phone number"
                      />
                      {isAddingMember && (
                        <p className="mt-1 text-xs text-blue-600 font-medium">Inherited from family account</p>
                      )}
                      {touched.phone && fieldErrors.phone && !isAddingMember && (
                        <p className="mt-1 text-xs text-red-600 font-medium">• {fieldErrors.phone}</p>
                      )}
                    </div>
                  </div>

                  <div className={`grid ${isAddingMember ? 'grid-cols-1' : 'grid-cols-2'} gap-4`}>
                    <div>
                      <label className={labelClass}>Relation *</label>
                      <select
                        value={formData.relation}
                        onChange={(e) => setFormData({ ...formData, relation: e.target.value })}
                        className={inputNormalClass}
                      >
                        {(isAddingMember ? ['Father', 'Mother', 'Son', 'Daughter', 'Wife', 'Husband', 'Other'] : RELATIONS).map((rel) => (
                          <option key={rel} value={rel}>{rel}</option>
                        ))}
                      </select>
                      {isAddingMember && (
                        <p className="mt-1 text-xs text-blue-600 font-medium">Relation to the family head</p>
                      )}
                    </div>
                    {!isAddingMember && (
                      <div>
                        <label className={labelClass}>Relation Detail</label>
                        <input
                          type="text"
                          value={formData.family_id || ''}
                          onChange={(e) => setFormData({ ...formData, family_id: e.target.value })}
                          className={inputNormalClass}
                          placeholder="Family linking (optional)"
                        />
                      </div>
                    )}
                  </div>
                </>
              )}

              {/* Visit-specific fields (shown for both new and returning patients) */}
              <div className="pt-2 border-t border-gray-200 mt-6">
                <h3 className="text-sm font-bold text-gray-700 uppercase tracking-wider mb-4">
                  {isReturningPatient ? '📝 Visit Details' : '📝 Initial Visit Details'}
                </h3>
              </div>

              <div>
                <label className={labelClass}>Basic Symptoms</label>
                <textarea
                  value={formData.symptoms}
                  onChange={(e) => setFormData({ ...formData, symptoms: e.target.value })}
                  className={inputNormalClass}
                  rows={3}
                  placeholder="Describe symptoms for this visit..."
                />
              </div>

              <div className="grid grid-cols-3 gap-4">
                <div>
                  <label className={labelClass}>🌡️ Temperature</label>
                  <input
                    type="text"
                    value={formData.temperature}
                    onChange={(e) => setFormData({ ...formData, temperature: e.target.value })}
                    className={inputNormalClass}
                    placeholder="e.g. 98.6°F"
                  />
                </div>
                <div>
                  <label className={labelClass}>🩸 Blood Pressure</label>
                  <input
                    type="text"
                    value={formData.blood_pressure}
                    onChange={(e) => setFormData({ ...formData, blood_pressure: e.target.value })}
                    className={inputNormalClass}
                    placeholder="e.g. 120/80"
                  />
                </div>
                <div>
                  <label className={labelClass}>💓 Heart Rate</label>
                  <input
                    type="text"
                    value={formData.heart_rate}
                    onChange={(e) => setFormData({ ...formData, heart_rate: e.target.value })}
                    className={inputNormalClass}
                    placeholder="e.g. 72 bpm"
                  />
                </div>
              </div>
              <div className="bg-blue-50 border-l-4 border-blue-500 p-3 text-xs text-blue-800">
                ℹ️ Medical vitals are optional. You can also fill these during doctor consultation.
              </div>

              <div>
                <label className={labelClass}>Assign Doctor *</label>
                <select
                  value={formData.doctor_id}
                  onChange={(e) => {
                    setFormData({ ...formData, doctor_id: e.target.value });
                    if (touched.doctor_id) setFieldErrors((p) => ({ ...p, doctor_id: validateField('doctor_id', e.target.value) }));
                  }}
                  onBlur={() => handleBlur('doctor_id')}
                  className={touched.doctor_id && fieldErrors.doctor_id ? inputErrorClass : inputNormalClass}
                >
                  <option value="">Select a doctor</option>
                  {doctors.map((doc) => (
                    <option key={doc.id} value={doc.id}>
                      {doc.user?.name} - {doc.specialization}
                    </option>
                  ))}
                </select>
                {touched.doctor_id && fieldErrors.doctor_id && (
                  <p className="mt-1 text-xs text-red-600 font-medium">• {fieldErrors.doctor_id}</p>
                )}
              </div>

              <div>
                <label className={labelClass}>Appointment Time</label>
                <input
                  type="datetime-local"
                  value={formData.appointment_time}
                  onChange={(e) => setFormData({ ...formData, appointment_time: e.target.value })}
                  className={inputNormalClass}
                />
              </div>

              <div className="bg-gray-50 border border-gray-200 p-3 text-xs text-gray-600">
                ℹ️ Next appointment (follow-up) will be set by the doctor during consultation.
              </div>

              <div className="flex justify-end pt-2">
                <button
                  type="submit"
                  disabled={submitting}
                  className={`px-8 py-3 text-white text-sm font-bold uppercase tracking-wider transition-colors duration-200 disabled:opacity-50 disabled:cursor-not-allowed flex items-center space-x-2 shadow-sm ${
                    isReturningPatient
                      ? 'bg-green-600 hover:bg-green-700'
                      : 'bg-blue-600 hover:bg-blue-700'
                  }`}
                >
                  {submitting && <Spinner />}
                  <span>
                    {submitting
                      ? 'Submitting...'
                      : isReturningPatient
                        ? '🔄 Create New Visit'
                        : isAddingMember
                          ? '➕ Add Family Member'
                          : '➕ Register Patient'
                    }
                  </span>
                </button>
              </div>
            </form>
          </div>
        )}

        {/* ── Patient List Tab ── */}
        {activeTab === 'list' && (
          <div className="space-y-4">
            <div className="flex flex-wrap gap-3 bg-white p-4 border border-gray-200 shadow-sm">
              <input
                type="text"
                placeholder="Search by name or phone..."
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                className={`${inputNormalClass} w-64`}
              />
              <select
                value={filterStatus}
                onChange={(e) => setFilterStatus(e.target.value)}
                className={`${inputNormalClass} w-48`}
              >
                <option value="">All Statuses</option>
                {Object.entries(STATUS_CONFIG).map(([key, config]) => (
                  <option key={key} value={key}>{config.label}</option>
                ))}
              </select>
            </div>

            <div className="bg-white border border-gray-200 shadow-sm overflow-hidden">
              <table className="w-full">
                <thead className="bg-gray-50 border-b border-gray-200">
                  <tr>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Name</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Age/Gender</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Phone</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Doctor</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Relation</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Status</th>
                    <th className="px-6 py-3 text-left text-xs font-bold uppercase tracking-wider text-gray-600">Visits</th>
                    <th className="px-6 py-3 text-right text-xs font-bold uppercase tracking-wider text-gray-600">Actions</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100">
                  {patients.map((patient) => (
                    <tr key={patient.id} className="hover:bg-blue-50 transition-colors duration-150">
                      <td className="px-6 py-4 text-sm font-bold text-gray-900">{patient.name}</td>
                      <td className="px-6 py-4 text-sm text-gray-600">{patient.age} / {patient.gender}</td>
                      <td className="px-6 py-4 text-sm text-gray-600">{patient.phone}</td>
                      <td className="px-6 py-4 text-sm text-gray-600">{patient.doctor?.user?.name || '-'}</td>
                      <td className="px-6 py-4">
                        <span className="px-2 py-1 text-xs font-semibold uppercase tracking-wider bg-gray-100 text-gray-700 border border-gray-200">
                          {patient.relation || 'Myself'}
                        </span>
                      </td>
                      <td className="px-6 py-4">
                        <StatusBadge status={patient.status} />
                      </td>
                      <td className="px-6 py-4 text-sm text-gray-600 font-medium">
                        {patient.consultations_count || 1}
                      </td>
                      <td className="px-6 py-4 text-right">
                        <button
                          onClick={() => handleWhatsAppDirect(patient)}
                          className="inline-flex items-center gap-1 px-4 py-2 bg-green-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-green-700 transition-colors duration-200 shadow-sm"
                          title="Send WhatsApp enquiry"
                        >
                          💬 WhatsApp
                        </button>
                      </td>
                    </tr>
                  ))}
                  {patients.length === 0 && (
                    <tr>
                      <td colSpan="8" className="px-6 py-16 text-center text-gray-500 font-medium">
                        No patients found
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* ── Scheduled Patients Tab ── */}
        {activeTab === 'schedule' && (
          <div className="space-y-4">
            <div className="flex items-center justify-between flex-wrap gap-3 bg-white p-4 border border-gray-200 shadow-sm">
              <div className="flex items-center gap-3">
                <label className="text-xs font-bold uppercase tracking-wider text-gray-600">📅 Date:</label>
                <input
                  type="date"
                  value={dateFilter}
                  onChange={(e) => setDateFilter(e.target.value)}
                  className={`${inputNormalClass} w-auto`}
                />
                <button
                  onClick={() => setDateFilter(new Date().toISOString().split('T')[0])}
                  className="px-4 py-2 text-xs font-bold uppercase tracking-wider bg-blue-50 text-blue-700 border border-blue-200 hover:bg-blue-100 transition-colors duration-200"
                >
                  Today
                </button>
              </div>
              <button
                onClick={fetchScheduledPatients}
                className="px-4 py-2 text-xs font-bold uppercase tracking-wider bg-gray-100 text-gray-700 border border-gray-200 hover:bg-gray-200 transition-colors duration-200 flex items-center gap-2"
              >
                🔄 Refresh <span className="normal-case font-normal text-gray-400">(auto: 30s)</span>
              </button>
            </div>

            {/* Summary cards */}
            <div className="grid grid-cols-3 gap-4">
              <div className="bg-white border border-gray-200 p-5 shadow-sm">
                <p className="text-xs font-bold uppercase tracking-wider text-gray-500">Total Scheduled</p>
                <p className="text-3xl font-bold text-gray-900 mt-2">{scheduledPatients.length}</p>
              </div>
              <div className="bg-white border border-gray-200 p-5 shadow-sm">
                <p className="text-xs font-bold uppercase tracking-wider text-gray-500">Pending</p>
                <p className="text-3xl font-bold text-blue-600 mt-2">
                  {scheduledPatients.filter((p) => p.status === 'pending' || p.status === 'assigned').length}
                </p>
              </div>
              <div className="bg-white border border-gray-200 p-5 shadow-sm">
                <p className="text-xs font-bold uppercase tracking-wider text-gray-500">Completed</p>
                <p className="text-3xl font-bold text-green-600 mt-2">
                  {scheduledPatients.filter((p) => p.status === 'completed').length}
                </p>
              </div>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
              {scheduledPatients.map((patient) => (
                <div key={patient.id} className="bg-white border border-gray-200 p-5 shadow-sm transition-all duration-200 hover:border-blue-400 hover:shadow-md flex flex-col">
                  <div className="flex items-start justify-between mb-4 border-b border-gray-100 pb-3">
                    <div>
                      <h3 className="font-bold text-gray-900 tracking-tight">{patient.name}</h3>
                      <p className="text-sm text-gray-500 font-medium">{patient.phone}</p>
                    </div>
                    <StatusBadge status={patient.status} />
                  </div>

                  <div className="space-y-2 text-sm flex-grow">
                    <div className="flex items-center justify-between">
                      <span className="text-gray-500 text-xs uppercase tracking-wider font-semibold">🕐 Appt</span>
                      <span className="font-bold text-gray-800">
                        {patient.consultation?.appointment_time
                          ? new Date(patient.consultation.appointment_time).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
                          : 'Not set'}
                      </span>
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-gray-500 text-xs uppercase tracking-wider font-semibold">👨‍⚕️ Doctor</span>
                      <span className="font-bold text-gray-800">{patient.doctor?.user?.name || '-'}</span>
                    </div>
                    {patient.consultation?.symptoms && (
                      <div className="pt-3 mt-3 border-t border-gray-100">
                        <p className="text-xs text-gray-500 uppercase tracking-wider font-semibold mb-1">Symptoms:</p>
                        <p className="text-sm text-gray-700 line-clamp-2 bg-gray-50 p-2 border border-gray-100">{patient.consultation.symptoms}</p>
                      </div>
                    )}
                  </div>

                  <div className="flex gap-2 mt-5">
                    <button
                      onClick={() => handleWhatsAppEnquiry(patient)}
                      className="flex-1 flex items-center justify-center gap-2 px-3 py-2.5 bg-green-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-green-700 transition-colors duration-200 shadow-sm"
                    >
                      <span>💬</span> WhatsApp
                    </button>
                    {patient.status !== 'completed' && (
                      <button
                        onClick={() => handleStatusChange(patient.id, 'completed')}
                        className="flex-1 px-3 py-2.5 bg-gray-100 text-gray-700 text-xs font-bold uppercase tracking-wider hover:bg-gray-200 transition-colors duration-200 border border-gray-200"
                      >
                        ✅ Complete
                      </button>
                    )}
                  </div>
                </div>
              ))}

              {scheduledPatients.length === 0 && (
                <div className="col-span-full bg-white border border-gray-200 p-16 text-center shadow-sm">
                  <p className="text-5xl mb-4">📅</p>
                  <p className="text-gray-600 font-bold text-lg tracking-tight">No patients scheduled for this date</p>
                </div>
              )}
            </div>
          </div>
        )}

        {/* ── Next Appointments Tab ── */}
        {activeTab === 'next' && (
          <div className="space-y-4">
            <div className="bg-blue-50 border-l-4 border-blue-600 p-4 text-sm text-blue-800 shadow-sm">
              ℹ️ These are follow-up appointments set by doctors during consultation. Only doctor-assigned follow-ups appear here.
            </div>

            <div className="flex items-center gap-3 bg-white p-4 border border-gray-200 shadow-sm">
              <label className="text-xs font-bold uppercase tracking-wider text-gray-600">📅 Filter by Date:</label>
              <input
                type="date"
                value={nextApptDate}
                onChange={(e) => setNextApptDate(e.target.value)}
                className={`${inputNormalClass} w-auto`}
              />
              <button
                onClick={() => setNextApptDate('')}
                className="px-4 py-2 text-xs font-bold uppercase tracking-wider bg-gray-100 text-gray-700 border border-gray-200 hover:bg-gray-200 transition-colors duration-200"
              >
                All Upcoming
              </button>
            </div>

            {loadingNext ? (
              <div className="bg-white border border-gray-200 p-16 text-center shadow-sm">
                <Spinner size="lg" />
                <p className="mt-4 text-gray-500 font-medium">Loading follow-ups...</p>
              </div>
            ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
              {nextAppointments.map((patient) => (
                <div key={patient.id} className="bg-white border border-gray-200 p-5 shadow-sm transition-all duration-200 hover:border-green-400 hover:shadow-md flex flex-col">
                  <div className="flex items-start justify-between mb-4 border-b border-gray-100 pb-3">
                    <div>
                      <h3 className="font-bold text-gray-900 tracking-tight">{patient.name}</h3>
                      <p className="text-sm text-gray-500 font-medium">{patient.phone}</p>
                    </div>
                    <StatusBadge status={patient.status} />
                  </div>

                  <div className="space-y-2 text-sm flex-grow">
                    <div className="flex items-center justify-between">
                      <span className="text-gray-500 text-xs uppercase tracking-wider font-semibold">📅 Follow-up</span>
                      <span className="font-bold text-blue-700">
                        {patient.consultation?.next_appointment
                          ? new Date(patient.consultation.next_appointment).toLocaleDateString()
                          : '-'}
                      </span>
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-gray-500 text-xs uppercase tracking-wider font-semibold">👨‍⚕️ Doctor</span>
                      <span className="font-bold text-gray-800">{patient.doctor?.user?.name || '-'}</span>
                    </div>
                    {patient.consultation?.diagnosis && (
                      <div className="pt-3 mt-3 border-t border-gray-100">
                        <p className="text-xs text-gray-500 uppercase tracking-wider font-semibold mb-1">Last Diagnosis:</p>
                        <p className="text-sm text-gray-700 line-clamp-2 bg-gray-50 p-2 border border-gray-100">{patient.consultation.diagnosis}</p>
                      </div>
                    )}
                  </div>

                  <div className="flex gap-2 mt-5">
                    <button
                      onClick={() => handleWhatsAppEnquiry(patient)}
                      className="w-full flex items-center justify-center gap-2 px-3 py-2.5 bg-green-600 text-white text-xs font-bold uppercase tracking-wider hover:bg-green-700 transition-colors duration-200 shadow-sm"
                    >
                      <span>💬</span> Send WhatsApp
                    </button>
                  </div>
                </div>
              ))}

              {nextAppointments.length === 0 && (
                <div className="col-span-full bg-white border border-gray-200 p-16 text-center shadow-sm">
                  <p className="text-5xl mb-4">🔮</p>
                  <p className="text-gray-600 font-bold text-lg tracking-tight">
                    {nextApptDate ? `No follow-ups scheduled for ${new Date(nextApptDate + 'T00:00:00').toLocaleDateString()}` : 'No follow-up appointments scheduled yet'}
                  </p>
                  <p className="text-xs text-gray-400 mt-2 font-medium uppercase tracking-wider">Follow-ups are set by doctors during consultation</p>
                </div>
              )}
            </div>
            )}
          </div>
        )}
      </main>
    </div>
  );
}
